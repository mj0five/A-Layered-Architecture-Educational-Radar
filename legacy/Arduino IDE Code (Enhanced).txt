/*
 * ============================================================================
 * Arduino-Based Ultrasonic Radar System
 * Dual-Mode Control: Automatic Sweep & Manual Joystick Targeting
 * ============================================================================
 * 
 * Hardware Platform: Arduino UNO R3 (ATmega328P)
 * Peripherals:
 *   - HC-SR04 Ultrasonic Distance Sensor (Time-of-Flight)
 *   - SG90 Micro Servomotor (PWM-controlled, 0-180°)
 *   - KY-023 Analog Joystick Module (X-axis + push-button switch)
 *   - KY-012 Active Piezoelectric Buzzer
 *   - KY-016 Common-Cathode RGB LED (PWM-driven)
 * 
 * Serial Protocol: CSV format [Angle],[Distance]. @ 9600 baud
 * 
 * Author: MohammadJavad Zaker Esfahani
 * Supervisor: Mehran Hosseinzadeh Dizaj
 * Institution: Islamic Azad University, Central Branch, Tehran, Iran
 * ============================================================================
 */

#include <Servo.h>

// =============================================================================
// PIN CONFIGURATION
// =============================================================================

// HC-SR04 Ultrasonic Sensor
const uint8_t PIN_TRIG = 10;        // Trigger output (digital)
const uint8_t PIN_ECHO = 11;        // Echo input (digital)

// SG90 Servomotor
const uint8_t PIN_SERVO = 9;        // PWM signal output

// KY-023 Joystick Module
const uint8_t PIN_JOYSTICK_X = A0;  // Analog X-axis input
const uint8_t PIN_JOYSTICK_SW = 2;  // Push-button switch (active-LOW)

// KY-012 Buzzer
const uint8_t PIN_BUZZER = 3;       // Digital output

// KY-016 RGB LED (Common Cathode)
const uint8_t PIN_LED_RED = 5;      // PWM channel - Red
const uint8_t PIN_LED_GREEN = 6;    // PWM channel - Green
const uint8_t PIN_LED_BLUE = 7;     // PWM channel - Blue

// =============================================================================
// OPERATIONAL PARAMETERS
// =============================================================================

// System Timing
const unsigned long SERIAL_BAUD = 9600;
const unsigned long MEASUREMENT_INTERVAL_MS = 50;   // Inter-sample delay
const unsigned long DEBOUNCE_DELAY_MS = 200;        // Switch debounce window
const unsigned long SERVO_DELAY_AUTO_MS = 40;       // Auto-mode step delay
const unsigned long SERVO_DELAY_MANUAL_MS = 20;     // Manual-mode update delay

// HC-SR04 Timing & Constraints
const unsigned long PULSE_TIMEOUT_US = 30000;       // 30 ms max echo wait
const float SOUND_SPEED_CM_US = 0.0343;             // Speed of sound (cm/us)
const uint16_t MAX_MEASURABLE_DISTANCE_CM = 400;    // Sensor absolute max
const uint16_t EFFECTIVE_RANGE_CM = 200;            // Reliable operational range
const uint16_t ALERT_THRESHOLD_CM = 50;             // Proximity alert trigger

// Servomotor Constraints
const uint8_t SERVO_MIN_ANGLE = 0;
const uint8_t SERVO_MAX_ANGLE = 180;
const uint8_t SERVO_CENTER_ANGLE = 90;
const int8_t SERVO_STEP_SIZE = 3;                   // Degrees per auto step

// Joystick Mapping
const uint16_t JOYSTICK_MIN_RAW = 0;
const uint16_t JOYSTICK_MAX_RAW = 1023;

// =============================================================================
// SYSTEM STATE VARIABLES
// =============================================================================

Servo radarServo;

// Operating mode: 0 = Automatic sweep, 1 = Manual joystick control
volatile uint8_t operatingMode = 0;

// Servo state
int16_t servoPosition = SERVO_CENTER_ANGLE;
int8_t servoDirection = 1;                          // +1: CW, -1: CCW

// Sensor state
uint16_t measuredDistance = 0;
unsigned long lastMeasurementTime = 0;

// Switch debounce state
uint8_t previousSwitchState = HIGH;
unsigned long lastDebounceTime = 0;

// =============================================================================
// INITIALIZATION
// =============================================================================

void setup() {
  // Configure digital I/O
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_JOYSTICK_SW, INPUT_PULLUP);   // Internal pull-up enabled
  pinMode(PIN_BUZZER, OUTPUT);
  pinMode(PIN_LED_RED, OUTPUT);
  pinMode(PIN_LED_GREEN, OUTPUT);
  pinMode(PIN_LED_BLUE, OUTPUT);
  
  // Initialize trigger line LOW
  digitalWrite(PIN_TRIG, LOW);
  
  // Initialize buzzer OFF
  digitalWrite(PIN_BUZZER, LOW);
  
  // Initialize LED to standby (blue)
  setRGBColor(0, 0, 255);
  
  // Attach and center servo
  radarServo.attach(PIN_SERVO);
  radarServo.write(servoPosition);
  delay(500);  // Allow servo to reach center position
  
  // Initialize serial communication for Processing visualization
  Serial.begin(SERIAL_BAUD);
  
  // System ready indication
  setRGBColor(0, 255, 0);  // Green: system ready
  delay(250);
  setRGBColor(0, 0, 0);    // Off
  delay(100);
  setRGBColor(0, 255, 0);
  delay(250);
  setRGBColor(0, 0, 0);
}

// =============================================================================
// MAIN CONTROL LOOP
// =============================================================================

void loop() {
  // --- Mode Switching Handler ---
  processModeSwitch();
  
  // --- Distance Measurement ---
  measuredDistance = measureDistance();
  
  // --- Alert Feedback Logic ---
  updateAlertSystem(measuredDistance);
  
  // --- Servo Control (Mode-Dependent) ---
  if (operatingMode == 0) {
    executeAutomaticSweep();
  } else {
    executeManualControl();
  }
  
  // --- Serial Data Transmission ---
  transmitTelemetry();
  
  // --- Loop Timing Control ---
  delay(MEASUREMENT_INTERVAL_MS);
}

// =============================================================================
// MODE SWITCHING WITH DEBOUNCE
// =============================================================================

void processModeSwitch(void) {
  uint8_t currentSwitchState = digitalRead(PIN_JOYSTICK_SW);
  
  // Detect falling edge (HIGH -> LOW) with debounce protection
  if ((currentSwitchState == LOW) && (previousSwitchState == HIGH)) {
    unsigned long currentTime = millis();
    
    if ((currentTime - lastDebounceTime) > DEBOUNCE_DELAY_MS) {
      operatingMode = !operatingMode;           // Toggle mode
      lastDebounceTime = currentTime;
      
      // Visual mode-change feedback
      setRGBColor(255, 255, 0);  // Yellow flash
      delay(150);
      
      // Reset servo to center on mode change for predictability
      servoPosition = SERVO_CENTER_ANGLE;
      radarServo.write(servoPosition);
    }
  }
  
  previousSwitchState = currentSwitchState;
}

// =============================================================================
// ULTRASONIC DISTANCE MEASUREMENT
// =============================================================================

uint16_t measureDistance(void) {
  // Transmit 10 us trigger pulse
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);
  
  // Measure echo pulse duration with timeout protection
  unsigned long pulseDuration = pulseIn(PIN_ECHO, HIGH, PULSE_TIMEOUT_US);
  
  // Handle timeout condition (no echo received)
  if (pulseDuration == 0) {
    return EFFECTIVE_RANGE_CM;  // Return max effective range as "no object"
  }
  
  // Calculate one-way distance: d = (t * v) / 2
  uint16_t calculatedDistance = (uint16_t)((pulseDuration * SOUND_SPEED_CM_US) / 2.0);
  
  // Constrain to valid operational envelope
  if (calculatedDistance > EFFECTIVE_RANGE_CM) {
    calculatedDistance = EFFECTIVE_RANGE_CM;
  }
  
  return calculatedDistance;
}

// =============================================================================
// ALERT & FEEDBACK SYSTEM
// =============================================================================

void updateAlertSystem(uint16_t distance) {
  if ((distance < ALERT_THRESHOLD_CM) && (distance > 0)) {
    // Target detected within alert zone
    digitalWrite(PIN_BUZZER, HIGH);
    setRGBColor(255, 0, 0);     // Red: alert state
  } else {
    // Clear field
    digitalWrite(PIN_BUZZER, LOW);
    setRGBColor(0, 255, 0);     // Green: all clear
  }
}

// =============================================================================
// SERVO CONTROL MODES
// =============================================================================

void executeAutomaticSweep(void) {
  // Increment position with directional step
  servoPosition += (servoDirection * SERVO_STEP_SIZE);
  
  // Boundary detection and direction reversal
  if (servoPosition >= SERVO_MAX_ANGLE) {
    servoPosition = SERVO_MAX_ANGLE;
    servoDirection = -1;
  } else if (servoPosition <= SERVO_MIN_ANGLE) {
    servoPosition = SERVO_MIN_ANGLE;
    servoDirection = 1;
  }
  
  // Apply position command
  radarServo.write((uint8_t)servoPosition);
  
  // Sweep smoothness delay
  delay(SERVO_DELAY_AUTO_MS);
}

void executeManualControl(void) {
  // Read joystick X-axis analog value
  uint16_t joystickValue = analogRead(PIN_JOYSTICK_X);
  
  // Map joystick range to servo angular range
  servoPosition = map(joystickValue, JOYSTICK_MIN_RAW, JOYSTICK_MAX_RAW, 
                      SERVO_MIN_ANGLE, SERVO_MAX_ANGLE);
  
  // Apply constrained position command
  servoPosition = constrain(servoPosition, SERVO_MIN_ANGLE, SERVO_MAX_ANGLE);
  radarServo.write((uint8_t)servoPosition);
  
  // Manual mode update delay
  delay(SERVO_DELAY_MANUAL_MS);
}

// =============================================================================
// SERIAL TELEMETRY PROTOCOL
// Format: [Angle],[Distance].
// Example: "90,45."
// =============================================================================

void transmitTelemetry(void) {
  Serial.print(servoPosition);
  Serial.print(",");
  Serial.print(measuredDistance);
  Serial.println(".");
}

// =============================================================================
// RGB LED COLOR CONTROL
// =============================================================================

void setRGBColor(uint8_t red, uint8_t green, uint8_t blue) {
  analogWrite(PIN_LED_RED, red);
  analogWrite(PIN_LED_GREEN, green);
  analogWrite(PIN_LED_BLUE, blue);
}
